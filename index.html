// =========================================================================
// Ultrasonic Glasses with Posture Correction & Caution Detection
// Target MCU: Arduino Nano (ATmega328P @ 16MHz)
// Sensors: HC-SR04 Ultrasound + MPU6050 6-DOF IMU
// Actuator: ERM Silent Coin Vibration Motor (PWM via 2N2222 transistor)
// =========================================================================

#include <Wire.h>

const int PIN_TRIG   = 8;   // 10us ultrasound trigger output
const int PIN_ECHO   = 7;   // Acoustic echo input interrupt
const int PIN_HAPTIC = 3;   // PWM gate to ERM motor driver

const float HAZARD_THRESHOLD_CM = 80.0;
const float POSTURE_SLOUCH_DEG  = 25.0;

void setup() {
  Serial.begin(115200);
  pinMode(PIN_TRIG, OUTPUT);
  pinMode(PIN_ECHO, INPUT);
  pinMode(PIN_HAPTIC, OUTPUT);
  digitalWrite(PIN_TRIG, LOW);

  Wire.begin();
  // Initialize MPU6050 over fast I2C bus
  Wire.beginTransmission(0x68);
  Wire.write(0x6B); // PWR_MGMT_1
  Wire.write(0x00); // Wake up device
  Wire.endTransmission(true);

  Serial.println(F("[SYSTEM] Ultrasonic Glasses Core Initialized."));
}

float measureDistanceCm() {
  digitalWrite(PIN_TRIG, LOW);
  delayMicroseconds(2);
  digitalWrite(PIN_TRIG, HIGH);
  delayMicroseconds(10);
  digitalWrite(PIN_TRIG, LOW);

  unsigned long duration = pulseIn(PIN_ECHO, HIGH, 30000); // 30ms timeout (~500cm)
  if (duration == 0) return 400.0;
  return (duration * 0.03432) / 2.0;
}

float measurePitchAngle() {
  Wire.beginTransmission(0x68);
  Wire.write(0x3B); // ACCEL_XOUT_H
  Wire.endTransmission(false);
  Wire.requestFrom(0x68, 6, true);

  int16_t ax = (Wire.read() << 8) | Wire.read();
  int16_t ay = (Wire.read() << 8) | Wire.read();
  int16_t az = (Wire.read() << 8) | Wire.read();

  // Pitch angle calculation from gravity vector
  float pitch = atan2((float)ay, sqrt((float)ax * ax + (float)az * az)) * 180.0 / PI;
  return abs(pitch);
}

void loop() {
  float dist = measureDistanceCm();
  float pitch = measurePitchAngle();

  bool obstacleHazard = (dist < HAZARD_THRESHOLD_CM);
  bool postureHazard  = (pitch > POSTURE_SLOUCH_DEG);

  if (obstacleHazard || postureHazard) {
    // Gentle pulse vibration warning
    analogWrite(PIN_HAPTIC, obstacleHazard ? 220 : 140);
    delay(80);
    analogWrite(PIN_HAPTIC, 0);
  }

  delay(30); // 30Hz evaluation loop
}
